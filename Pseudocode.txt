PSEUDOCODE


TimesAndDurationCalculator


BEGIN


    DECLARE StartAgain AS STRING
    DECLARE StartHour, StartMinute, EndHour, EndMinute AS INTEGER
    DECLARE HourDuration, MinuteDuration AS INTEGER


    SET StartAgain = "yes"


    WHILE StartAgain = "yes" DO


        INPUT StartHour PROMPT "Enter 2-digit start hour (00-23): "
        INPUT StartMinute PROMPT "Enter 2-digit start minute (00-59): "
        INPUT EndHour PROMPT "Enter 2-digit end hour (00-23): "
        INPUT EndMinute PROMPT "Enter 2-digit end minute (00-59): "


        IF EndHour < StartHour THEN
            HourDuration = EndHour + 24 - StartHour
        ELSE
            HourDuration = EndHour - StartHour
        END IF


        IF EndMinute < StartMinute THEN
            MinuteDuration = EndMinute + 60 - StartMinute
            HourDuration = HourDuration - 1
        ELSE
            MinuteDuration = EndMinute - StartMinute
        END IF


        PRINT ""
        PRINT "Start: " + StartHour + ":" + StartMinute
        PRINT "End: " + EndHour + ":" + EndMinute
        PRINT "Duration: " + HourDuration + " hour(s) " + MinuteDuration + " minute(s)"


        PRINT ""
        PRINT "Do you want to perform another calculation? (yes/no)"
        INPUT StartAgain
        PRINT ""


    END WHILE


    PRINT "Thank you for using our calculator."


END
